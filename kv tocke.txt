23. Sortiranje – preporuka koristiti ugrađenu qsort() funkciju, inače se može koristiti bilo koja
funkcija sortiranja.

qsort(lista, broj, sizeof(Valuta), usporedbaTecajeva);

int usporedbaTecajeva(const void *a, const void *b) {
    const Valuta *v1 = a;
    const Valuta *v2 = b;
    return (v1->tecaj > v2->tecaj) - (v1->tecaj < v2->tecaj);
}

24. Pretraživanje – preporuka koristiti ugrađenu bsearch() funkciju, inače se može koristiti bilo koja
funkcija pretraživanja.

int rekurzivnoTrazi(Valuta *lista, int broj, const char *kodValute) {
    if (broj == 0) return -1;
    if (strcmp(lista[broj - 1].kod, kodValute) == 0)
        return broj - 1;
    return rekurzivnoTrazi(lista, broj - 1, kodValute);
}
